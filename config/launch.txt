#Making package:
mkdir -p robot_ws/src
cd src
git clone https://github.com/NaviX-1/my_bot.git
cd my_bot
git checkout robot_roscontrol
cd ..
git clone https://github.com/joshnewans/serial_motor_demo.git
git clone https://github.com/joshnewans/serial.git
git clone https://github.com/joshnewans/diffdrive_arduino.git
cd ~/robot_ws
colcon build --symlink-install


#Launch Gazebo with NaviX                                                            
Source the work space: source ~/robot_ws/install/setup.bash                                              
Launch Gazebo: ros2 launch my_bot launch_robot.launch.py                                              
Control NaviX: ros2 run teleop_twist_keyboard teleop_twist_keyboard --ros-args -r /cmd_vel:=/diff_cont/cmd_vel_unstamped
